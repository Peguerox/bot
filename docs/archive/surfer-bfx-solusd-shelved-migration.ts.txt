// Surfer USDT on Bitfinex — same RSI(14)/EMA(7/25)/trailing-stop formula as
// trigger/live-bot-surfer-solusdt.ts (the real-money Binance.US version), ported to Bitfinex
// execution. Found 2026-09-15/16: real trade history vs backtest on the exact same window showed
// a ~26-hour entry-timing lag on one trade alone costing ~10.5pp (Aug 17-18), traced to Trigger.dev's
// ~50s scheduled-task dispatch delay stacked on the 5-minute cron, plus the old limit-chase
// execution taking further ticks to fill. This version checks every 30s (matches the strategy's own
// finest interval, 15m candles) and executes with a real Bitfinex EXCHANGE MARKET order over the
// authenticated WS -- instant fill, no chase loop. NOT ENABLED -- state starts enabled=false. Runs
// in the same process as server/surfer-bfx-solbtc.ts.
import { getBitfinexCandlesOHLCV } from "../lib/bitfinex";
import { getBookBidAsk } from "../lib/bitfinex-multi-book-ws";
import { submitMarketOrderFast, getLiveBalance, isWalletReady } from "../lib/bitfinex-trading-ws";
import {
  getSurferBfxSolusdState, updateSurferBfxSolusdState, recordSurferBfxSolusdTrade,
  logSurferBfxSolusdRun, type SurferBfxSolusdState,
} from "../lib/surfer-bfx-solusd-db";

const SYMBOL         = "tSOLUSD";
const RSI_LOW        = 30;
const MA_FAST        = 7;
const MA_SLOW        = 25;
const C15_LIMIT       = 110;
const C12H_LIMIT      = 100;
const CHECK_INTERVAL_MS = 30_000;
const RUN_LOG_INTERVAL_MS = 5 * 60_000;

const HARD_STOP_PCT   = -6;
const TRAIL_ARM_PCT   = 8;
const TRAIL_PP        = 10;
const STEP_THRESH     = 30;
const STEP_TRAIL_PP   = 6;

type Candle = { time: number; close: number };

function calcEMA(candles: Candle[], period: number): number[] {
  const k = 2 / (period + 1);
  const out: number[] = new Array(candles.length).fill(NaN);
  if (candles.length < period) return out;
  out[period - 1] = candles.slice(0, period).reduce((s, c) => s + c.close, 0) / period;
  for (let i = period; i < candles.length; i++) {
    out[i] = candles[i].close * k + out[i - 1] * (1 - k);
  }
  return out;
}

function calcRSI(candles: Candle[], period = 14): number[] {
  const rsi: number[] = new Array(candles.length).fill(NaN);
  if (candles.length < period + 1) return rsi;
  let avgGain = 0, avgLoss = 0;
  for (let i = 1; i <= period; i++) {
    const d = candles[i].close - candles[i - 1].close;
    if (d > 0) avgGain += d; else avgLoss += -d;
  }
  avgGain /= period; avgLoss /= period;
  rsi[period] = avgLoss === 0 ? 100 : 100 - 100 / (1 + avgGain / avgLoss);
  for (let i = period + 1; i < candles.length; i++) {
    const d = candles[i].close - candles[i - 1].close;
    avgGain = (avgGain * (period - 1) + (d > 0 ? d : 0)) / period;
    avgLoss = (avgLoss * (period - 1) + (d < 0 ? -d : 0)) / period;
    rsi[i] = avgLoss === 0 ? 100 : 100 - 100 / (1 + avgGain / avgLoss);
  }
  return rsi;
}

let lastRunLog = 0;

async function checkOnce() {
  let state: SurferBfxSolusdState;
  try {
    state = await getSurferBfxSolusdState();
  } catch (err) {
    await logSurferBfxSolusdRun({ actions: [{ action: "ERROR", stage: "state", error: String(err) }] }).catch(() => {});
    return;
  }
  if (!state.enabled) return;
  if (!isWalletReady()) return;

  const { bid, ask } = getBookBidAsk(SYMBOL);
  if (bid === null || ask === null) return;
  const livePrice = (bid + ask) / 2;

  try {
    const [raw15, raw12h] = await Promise.all([
      getBitfinexCandlesOHLCV(SYMBOL, "15m", C15_LIMIT),
      getBitfinexCandlesOHLCV(SYMBOL, "12h", C12H_LIMIT),
    ]);
    const c15: Candle[] = raw15.slice(0, -1).map((c) => ({ time: c.time, close: c.close }));
    const c12h: Candle[] = raw12h.slice(0, -1).map((c) => ({ time: c.time, close: c.close }));

    const rsiArr = calcRSI(c15);
    const curRSI = rsiArr[rsiArr.length - 1];
    const prevRSI = rsiArr[rsiArr.length - 2];
    const lastCandleTs = c15[c15.length - 1].time;

    const ema7Arr = calcEMA(c12h, MA_FAST);
    const ema25Arr = calcEMA(c12h, MA_SLOW);
    const lastEma7 = ema7Arr[ema7Arr.length - 1];
    const prevEma7 = ema7Arr[ema7Arr.length - 2];
    const lastEma25 = ema25Arr[ema25Arr.length - 1];
    const lastClose12h = c12h[c12h.length - 1].close;
    const delta = livePrice - lastClose12h;
    const liveEma7 = lastEma7 + delta / MA_FAST;
    const liveEma25 = lastEma25 + delta / MA_SLOW;
    const emaBullish = !isNaN(liveEma7) && !isNaN(liveEma25) && liveEma7 > liveEma25;
    const emaSloping = !isNaN(prevEma7) && liveEma7 > prevEma7;

    let bestPct = state.best_pct ?? 0;
    let curPct = 0;
    let hardStopHit = false;
    let trailHit = false;
    if (state.mode === "SOL" && state.entry_price) {
      curPct = (livePrice - state.entry_price) / state.entry_price * 100;
      if (curPct > bestPct) { bestPct = curPct; await updateSurferBfxSolusdState({ best_pct: bestPct }); }
      if (curPct <= HARD_STOP_PCT) hardStopHit = true;
      else if (bestPct >= TRAIL_ARM_PCT) {
        const trailPp = bestPct >= STEP_THRESH ? STEP_TRAIL_PP : TRAIL_PP;
        if (bestPct - curPct >= trailPp) trailHit = true;
      }
    }

    const shouldLog = Date.now() - lastRunLog > RUN_LOG_INTERVAL_MS;
    if (shouldLog) {
      await logSurferBfxSolusdRun({
        actions: [{ action: "CHECK", mode: state.mode, rsi: curRSI?.toFixed(2), emaBullish, emaSloping, price: livePrice, armed_sol: state.armed_for_sol, curPct: curPct.toFixed(2), bestPct: bestPct.toFixed(2) }],
      });
      lastRunLog = Date.now();
    }

    let armedForSol = state.armed_for_sol;
    const isNewCandle = lastCandleTs > (state.last_candle_ts ?? 0);
    if (isNewCandle && !isNaN(curRSI) && !isNaN(prevRSI)) {
      const candleUpdates: Record<string, unknown> = { last_candle_ts: lastCandleTs };
      if (state.mode === "USD" && prevRSI < RSI_LOW && curRSI >= RSI_LOW && !armedForSol) {
        armedForSol = true;
        candleUpdates.armed_for_sol = true;
        await logSurferBfxSolusdRun({ actions: [{ action: "ARM_BUY", prevRSI: prevRSI.toFixed(2), curRSI: curRSI.toFixed(2) }] });
      }
      await updateSurferBfxSolusdState(candleUpdates);
    }

    // ── Fire buy ──────────────────────────────────────────────────────
    if (state.mode === "USD" && armedForSol && emaBullish && emaSloping) {
      const usdFree = getLiveBalance("USD");
      const usdToUse = Math.min(usdFree, state.usd_balance);
      if (usdToUse > 0) {
        const qty = usdToUse / ask;
        console.log(`SOLUSD ENTRY armed+bullish+sloping usdToUse=${usdToUse.toFixed(2)} — submitting real market order...`);
        const fill = await submitMarketOrderFast(SYMBOL, qty);
        const usdSpent = fill.execPrice * Math.abs(fill.execAmount);
        await updateSurferBfxSolusdState({
          mode: "SOL", sol_quantity: Math.abs(fill.execAmount), entry_price: fill.execPrice,
          entry_usd: usdSpent, entry_time: new Date().toISOString(), armed_for_sol: false, best_pct: 0,
        });
        await logSurferBfxSolusdRun({ actions: [{ action: "BUY_FILLED", price: fill.execPrice, qty: fill.execAmount, usdSpent, fillLatencyMs: fill.latencyMs }] });
      }
    }

    // ── Fire sell ─────────────────────────────────────────────────────
    const trendExit = !emaBullish && !isNaN(curRSI) && curRSI < 50;
    if (state.mode === "SOL" && (hardStopHit || trailHit || trendExit)) {
      const trackedQty = state.sol_quantity ?? 0;
      const solFree = getLiveBalance("SOL");
      const sellQty = Math.min(trackedQty, solFree);
      if (sellQty > 0) {
        const reason = hardStopHit ? "hardstop" : trailHit ? "trail" : "trend";
        console.log(`SOLUSD EXIT reason=${reason} sellQty=${sellQty.toFixed(6)} — submitting real market order...`);
        const fill = await submitMarketOrderFast(SYMBOL, -sellQty);
        const usdOut = fill.execPrice * Math.abs(fill.execAmount);
        const usdIn = state.entry_usd ?? 0;
        const pnlUsd = usdOut - usdIn;
        const pnlPct = usdIn > 0 ? (pnlUsd / usdIn) * 100 : 0;
        await recordSurferBfxSolusdTrade({
          entry_price: state.entry_price!, exit_price: fill.execPrice, sol_quantity: state.sol_quantity!,
          usd_in: usdIn, usd_out: usdOut, pnl_usd: pnlUsd, pnl_pct: pnlPct, entry_time: state.entry_time!,
        });
        await updateSurferBfxSolusdState({
          mode: "USD", sol_quantity: null, entry_price: null, entry_usd: null, entry_time: null,
          usd_balance: usdOut, best_pct: 0,
        });
        await logSurferBfxSolusdRun({ actions: [{ action: "SELL_FILLED", reason, price: fill.execPrice, usdOut, pnlUsd, pnlPct, fillLatencyMs: fill.latencyMs }] });
      }
    }
  } catch (err) {
    console.error("Surfer BFX SOLUSD checkOnce error:", err);
    await logSurferBfxSolusdRun({ actions: [{ action: "ERROR", stage: "check", error: String(err) }] }).catch(() => {});
  }
}

export function startSurferBfxSolusd() {
  console.log("Surfer USDT (Bitfinex SOLUSD) engine starting — checking every 30s, NOT ENABLED until state.enabled=true.");
  checkOnce().catch((err) => console.error(err));
  setInterval(() => { checkOnce().catch((err) => console.error(err)); }, CHECK_INTERVAL_MS);
}

export async function surferBfxSolusdHeartbeat(instanceId: string) {
  await updateSurferBfxSolusdState({ lock_owner: instanceId, lock_heartbeat: new Date().toISOString() });
}
